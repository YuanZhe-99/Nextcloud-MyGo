  load_cache("${SYNCD_UPSTREAM_BUILD}" READ_WITH_PREFIX UPSTREAM_ CMAKE_HOME_DIRECTORY SQLite3_INCLUDE_DIR OPENSC-LIBP11_INCLUDE_DIRS)
  find_package(Qt6 6.12 REQUIRED COMPONENTS Gui Xml WebSockets)
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/libsync/syncengine.cpp" upstream_engine_source)
  string(FIND "${upstream_engine_source}" "Never create a propagator after that terminal transition" veto_marker)
  if(veto_marker LESS 0)
    message(FATAL_ERROR "SyncD requires the recorded pre-propagation veto patch")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/common/syncjournaldb.h" upstream_journal_header)
  string(FIND "${upstream_journal_header}" "hasCommitError" commit_marker)

    message(FATAL_ERROR "SyncD requires the recorded observable journal-commit patch")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/common/syncjournaldb.cpp" upstream_journal_source)
  string(FIND "${upstream_journal_source}" "openOrCreateReadWrite(_dbFile, false)" preserve_marker)
  if(preserve_marker LESS 0)
    message(FATAL_ERROR "SyncD requires the recorded journal-preservation patch")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/libsync/account.h" upstream_account_header)
  string(FIND "${upstream_account_header}" "create(bool withUserStatus" headless_marker)
  if(headless_marker LESS 0)
    message(FATAL_ERROR "SyncD requires the recorded headless-account lifetime patch")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/libsync/account.cpp" upstream_account_source)
  string(FIND "${upstream_account_source}" "acc->_headless = !withUserStatus;" headless_capability_marker)
  if(headless_capability_marker LESS 0)
    message(FATAL_ERROR "SyncD requires headless capability updates without background services")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/libsync/propagatedownload.cpp" upstream_download_source)
  string(FIND "${upstream_download_source}" "Resume state is keyed by _file when the transfer starts" download_resume_marker)
  if(download_resume_marker LESS 0)
    message(FATAL_ERROR "SyncD requires the recorded completed-download resume cleanup patch")
  endif()
  file(READ "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/common/vfs.h" upstream_vfs_header)
  string(FIND "${upstream_vfs_header}" "syncRootAccountId" cfapi_marker)
  set(SYNCD_CFAPI_PLUGIN "${SYNCD_UPSTREAM_BUILD}/bin/Release/nextcloudsync_vfs_cfapi.dll")
  if(cfapi_marker LESS 0 OR NOT EXISTS "${SYNCD_CFAPI_PLUGIN}")
    message(FATAL_ERROR "SyncD requires the recorded CFAPI embedding patch and its built plugin")
  endif()
  target_sources(syncd_core PRIVATE engine.cpp engine_runtime.cpp platform/windows/ondemand_windows.cpp)
  target_compile_definitions(syncd_core PUBLIC SYNCD_WITH_ENGINE)
  target_compile_definitions(syncd_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  set(SYNCD_UPSTREAM_INCLUDES
    "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src" "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/libsync"
    "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/csync" "${UPSTREAM_CMAKE_HOME_DIRECTORY}/src/common"
    "${SYNCD_UPSTREAM_BUILD}" "${SYNCD_UPSTREAM_BUILD}/src" "${SYNCD_UPSTREAM_BUILD}/src/libsync"
    "${SYNCD_UPSTREAM_BUILD}/src/csync" "${SYNCD_UPSTREAM_BUILD}/src/common" "${UPSTREAM_SQLite3_INCLUDE_DIR}" "${UPSTREAM_OPENSC-LIBP11_INCLUDE_DIRS}"
    "${SYNCD_UPSTREAM_BUILD}/src/libsync/vfs/cfapi")

  target_link_libraries(syncd_core PUBLIC Qt6::Gui Qt6::Xml Qt6::WebSockets
    "${SYNCD_UPSTREAM_BUILD}/lib/Release/nextcloudsync.lib"

  target_include_directories(syncd_ondemand_acceptance PRIVATE ${SYNCD_UPSTREAM_INCLUDES})

# Every test finds the exact DLLs this build linked against, whatever PATH its
# caller has: Qt, the recorded upstream build and the native dependencies, the
# same directories tools/test-all.ps1 uses. A stale PATH (an older upstream
# tree, or Qt only) otherwise fails the loader with 0xc0000135.
if(BUILD_TESTING)
  get_property(syncd_tests DIRECTORY PROPERTY TESTS)
  if(syncd_tests)
    if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
      set(syncd_native arm64)
    else()
      set(syncd_native x64)
    endif()
    get_filename_component(syncd_dev "${CMAKE_CURRENT_SOURCE_DIR}/../.dev" ABSOLUTE)
    set(syncd_runtime "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_BINS}")
    if(SYNCD_UPSTREAM_BUILD)
      list(APPEND syncd_runtime "${SYNCD_UPSTREAM_BUILD}/bin/Release")
    endif()
    list(APPEND syncd_runtime "${syncd_dev}/native/${syncd_native}/bin"
         "${syncd_dev}/vcpkg/installed/${syncd_native}-windows/bin")
    list(REVERSE syncd_runtime)
    set(syncd_environment
        "QT_PLUGIN_PATH=set:${QT6_INSTALL_PREFIX}/${QT6_INSTALL_PLUGINS}")
    foreach(directory IN LISTS syncd_runtime)
      file(TO_NATIVE_PATH "${directory}" directory)
      list(APPEND syncd_environment "PATH=path_list_prepend:${directory}")
    endforeach()
    set_tests_properties(${syncd_tests} PROPERTIES ENVIRONMENT_MODIFICATION
                                                   "${syncd_environment}")
  endif()
endif()
